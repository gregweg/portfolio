# schemas/member_schema.py

from pydantic import BaseModel
from typing import Literal


class Member(BaseModel):
    id: str
    member_id: str
    first_name: str
    last_name: str
    plan_name: str
    status: Literal["ACTIVE", "INACTIVE"]


class MemberSearchResponse(BaseModel):
    members: list[Member]


class MemberResponse(BaseModel):
    member: Member

# repositories/member_repository.py

from schemas.member_schema import Member


class MemberRepository:
    def __init__(self) -> None:
        self.members = [
            Member(
                id="1",
                member_id="ABC123",
                first_name="Greg",
                last_name="Wegman",
                plan_name="Blue Shield PPO",
                status="ACTIVE",
            ),
            Member(
                id="2",
                member_id="XYZ789",
                first_name="Jane",
                last_name="Smith",
                plan_name="Anthem HMO",
                status="INACTIVE",
            ),
        ]

    def search(self, query: str) -> list[Member]:
        query_lower = query.lower()

        return [
            member
            for member in self.members
            if query_lower in member.member_id.lower()
            or query_lower in member.first_name.lower()
            or query_lower in member.last_name.lower()
        ]

    def find_by_id(self, member_id: str) -> Member | None:
        return next(
            (member for member in self.members if member.id == member_id),
            None,
        )
# services/member_service.py

from repositories.member_repository import MemberRepository
from schemas.member_schema import Member


class MemberService:
    def __init__(self, member_repository: MemberRepository) -> None:
        self.member_repository = member_repository

    def search_members(self, query: str) -> list[Member]:
        return self.member_repository.search(query)

    def get_member_by_id(self, member_id: str) -> Member | None:
        return self.member_repository.find_by_id(member_id)

# routers/member_router.py

from fastapi import APIRouter, Depends, HTTPException, Query

from repositories.member_repository import MemberRepository
from schemas.member_schema import MemberResponse, MemberSearchResponse
from services.member_service import MemberService


router = APIRouter(prefix="/api/members", tags=["members"])


def get_member_repository() -> MemberRepository:
    return MemberRepository()


def get_member_service(
    repository: MemberRepository = Depends(get_member_repository),
) -> MemberService:
    return MemberService(repository)


@router.get("", response_model=MemberSearchResponse)
def search_members(
    query: str = Query(..., min_length=2),
    member_service: MemberService = Depends(get_member_service),
):
    members = member_service.search_members(query)

    return MemberSearchResponse(members=members)


@router.get("/{member_id}", response_model=MemberResponse)
def get_member_by_id(
    member_id: str,
    member_service: MemberService = Depends(get_member_service),
):
    member = member_service.get_member_by_id(member_id)

    if member is None:
        raise HTTPException(
            status_code=404,
            detail="Member not found",
        )

    return MemberResponse(member=member)

# main.py

from fastapi import FastAPI

from routers.member_router import router as member_router


app = FastAPI(title="Member Search API")

app.include_router(member_router)
