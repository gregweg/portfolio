Variable Length

class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        l, longest = 0, 0
        sett = set()
        n =  len(s)

        for r in range(n): # O(n)
            while s[r] in sett: # 
                sett.remove(s[l])
                l += 1
            w = (r - l) + 1
            longest = max(longest, w)
            sett.add(s[r])
        return longest
        # Time: O(n)
        # Space: O(n)

Fixed Length

def findMaxAverage(self, nums: List[int], k: int) -> float:
    n = len(nums)
    cur_sum = 0

    for i in range(k):
        cur_sum += nums[i]

    max_avg = cur_sum / k

    for i in range(k, n):
        cur_sum += nums[i]
        cur_sum -= nums[i-k]

        max_avg = max(max_avg, cur_sum / k)
    return max_avg
    # Time: O(n)
    # Space: O(1)
# nums = [1,12,-5,-6,50,3], k = 4
#findMaxAverage(nums, k) -> 12.75

# Max Heap

pq = []

# Goal: Lowest Tier first, Highest Score second
# We negate the Score so the highest score becomes the "smallest" negative number
tasks = [
    (1, 95, "User A"),
    (2, 99, "User B"),
    (1, 100, "User C")
]

for tier, score, user in tasks:
    # Notice the negative sign on score
    heapq.heappush(pq, (tier, -score, user))

while pq:
    tier, neg_score, user = heapq.heappop(pq)
    print(f"Tier: {tier}, Score: {-neg_score}, User: {user}")


# To avoid the uncomparable data trap, use dataclasses


@dataclass(order=True)
class PrioritizedTask:
    primary: int
    secondary: int
    # compare=False forces Python to ignore this field during heap operations
    data: dict = field(compare=False) 

pq = []
heapq.heappush(pq, PrioritizedTask(1, 5, {"id": "task_1"}))
heapq.heappush(pq, PrioritizedTask(1, 5, {"id": "task_2"})) # Identical priorities, no TypeError

# Safely pop
highest_priority_item = heapq.heappop(pq)
print(highest_priority_item.data)

Binary Search Iterative

def binary_search_iterative(arr, target):
    left = 0
    right = len(arr) - 1

    while left <= right:
        mid = left + (right - left) // 2

        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1 # target larger, ignore left half
        else:
            right = mid - 1 # target smaller, ignore right half
    return -1 # target not present in list

LRU Cache

from collections import OrderedDict

class LRUCache:
  def __init__(self, n):
    self.cache = OrderedDict()
    self.capacity = n

  def get(self, key):
    if key not in self.cache:
      return None

    self.cache.move_to_end(key)
    return self.cache[key]

  def set(self, key, val):
    if key in self.cache:
      self.cache.move_to_end(key)
    self.cache[key] = val

    if len(self.cache) > self.capacity:
      self.cache.popitem(last=False)

# debug your code below
cache = LRUCache(2)
cache.set('user1', 'Alex')
print(cache.get('user1'))
