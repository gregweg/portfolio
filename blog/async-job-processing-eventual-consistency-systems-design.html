from datetime import datetime
from typing import List, Dict, Any, Optional
from pydantic import BaseModel, Field, field_validator

class AgentStep(BaseModel):
    step_id: str = Field(..., description="Unique UUID for the trajectory step")
    tool_called: str = Field(..., min_length=1)
    execution_timestamp: datetime
    output_payload: Dict[str, Any]
    latency_ms: int

    @field_validator("latency_ms")
    @classmethod
    def validate_latency(cls, value: int) -> int:
        if value < 0:
            raise ValueError("Latency cannot be negative")
        return value

class EvaluationCriteria(BaseModel):
    required_tools: List[str]
    max_total_latency_ms: int
    expected_output_keys: List[str]

class TrajectoryEvaluationReport(BaseModel):
    trajectory_id: str
    is_successful: bool
    final_score: float
    tools_utilized: List[str]
    anomalies_detected: List[str]

    def save_evaluation_report(self, report: TrajectoryEvaluationReport) -> bool:
        """Persists evaluation telemetry to downstream DB layers with simple retry."""
        for attempt in range(self.max_retries):
            try:
                # In real life, this targets a Postgres connection pool or an internal API endpoint
                print(f"[I/O] Successfully saved trajectory report {report.trajectory_id}")
                return True
            except Exception:
                if attempt == self.max_retries - 1:
                    raise RuntimeError("Downstream operational layer unavailable.")
                time.sleep(0.1)
        return False

# LAYER 3: CORE EVALUATION BUSINESS LOGIC
# ==========================================
class AgentTrajectoryVerifier:
    """
    Evaluates whether an agent's execution sequence successfully satisfied 
    the targeted benchmark evaluation requirements.
    """
    # Explicit configurations replacing hidden magic numbers
    PERFECT_SCORE_PERCENTAGE = 100.0
    LATENCY_PENALTY_FACTOR = 0.15
    MINIMUM_VIABLE_SCORE = 70.0

    def __init__(self, criteria: EvaluationCriteria):
        self.criteria = criteria

    def evaluate_trajectory(self, trajectory_id: str, steps: List[AgentStep]) -> TrajectoryEvaluationReport:
        if not steps:
            return TrajectoryEvaluationReport(
                trajectory_id=trajectory_id,
                is_successful=False,
                final_score=0.0,
                tools_utilized=[],
                anomalies_detected=["Empty execution trajectory sequence."]
            )

        # 1. Transform arrays to sets for fast, O(1) membership evaluations
        unique_tools_used = {step.tool_called for step in steps}
        required_tools_set = set(self.criteria.required_tools)

        # 2. Extract metrics using clear helper calculations
        total_latency = sum(step.latency_ms for step in steps)
        anomalies = self._detect_anomalies(steps, total_latency)

        # 3. Score calculation based on criteria requirements
        score = self._calculate_trajectory_score(unique_tools_used, required_tools_set, total_latency)

        # Determine strict success criteria bounds
        meets_score_bar = score >= self.MINIMUM_VIABLE_SCORE
        has_critical_failures = len(anomalies) > 0 and total_latency > self.criteria.max_total_latency_ms

        return TrajectoryEvaluationReport(
            trajectory_id=trajectory_id,
            is_successful=meets_score_bar and not has_critical_failures,
            final_score=round(score, 2),
            tools_utilized=list(unique_tools_used),
            anomalies_detected=anomalies
        )

    def _detect_anomalies(self, steps: List[AgentStep], total_latency: int) -> List[str]:
        """Helper function focused strictly on structural trajectory telemetry faults."""
        detected_faults = []
        if total_latency > self.criteria.max_total_latency_ms:
            detected_faults.append(f"Total latency exceeded threshold by {total_latency - self.criteria.max_total_latency_ms}ms")

        # Track out-of-order timelines
        for index in range(len(steps) - 1):
            if steps[index].execution_timestamp > steps[index + 1].execution_timestamp:
                detected_faults.append(f"Chronological anomaly detected at step index {index}")
                break

        return detected_faults

    def _calculate_trajectory_score(self, tools_used: set, required_tools: set, total_latency: int) -> float:
        """Helper function computing performance scores against target criteria metrics."""
        if not required_tools:
            return self.PERFECT_SCORE_PERCENTAGE

        matched_tools = tools_used.intersection(required_tools)
        coverage_ratio = len(matched_tools) / len(required_tools)
        base_score = coverage_ratio * self.PERFECT_SCORE_PERCENTAGE

        # Apply structural deduction penalty if execution is sluggish
        if total_latency > self.criteria.max_total_latency_ms:
            base_score -= (base_score * self.LATENCY_PENALTY_FACTOR)

        return max(0.0, base_score)

import time
import random

def retry_with_backoff(fn, max_attempts=5):
    delay = 1

    for attempt in range(max_attempts):
        try:
            return fn()
        except RateLimitError:
            jitter = random.uniform(0, 0.25)
            time.sleep(delay + jitter)
            delay *= 2

    raise Exception("Max retries exceeded")
