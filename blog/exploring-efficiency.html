Course Schedule

Directed Graphs can be used to represent courses with edges representing the prerequisite relationships. Below we employ a queue in a BFS manner to keep track of nodes with zero in-degrees and thus no remanining prerequisites. Using a list to track the number of prerequisites for each course.

Each course with no in-degrees or prerequisites remaining is dequeued and it's neighbors are processed for their reduced in-degrees. When 0 in-degrees added to the queue.

Keeping track of processed courses we are able to see if all courses can be finished. If the count of processed courses matches total number of courses there are no cyclic dependencies and the courses can be completed.

from collections import deque, defaultdict
from typing import List

def can_finish(num_courses: int, prerequisites: List[List[int]]) -> bool:
    # Adjacency List and In-Degree count
    adjacency_list = defaultdict(list)
    in_degree = [0] * num_courses

    # Build the graph
    for dest, src in prerequisites:
        adjacency_list[src].append(dest)
        in_degree[dest] += 1

    # Initialize a queue with all courses having zero in-degree
    queue = deque([i for i in range(num_courses) if in_degree[i] == 0])

    visited_courses = 0

    # Process nodes with zero in-degree, potential starting course.
    while queue:
        node = queue.popleft()
        visited_courses += 1
        for neighbor in adjacency_list[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    # Check if we visited all courses
    return visited_courses == num_courses

print(can_finish(4, [[1, 0], [2, 1], [3, 2]]))

Determine Maximum Frequency and Task count, along with how many similar have this maximum frequency.

Calculate required idle slots needed between the most frequent tasks to ensure they are spaced out according to the cooldown period.

def least_interval_naive(tasks: List[str], n: int) -> int:
    # count number of each tasks
    task_counts = Counter(tasks)
    interval = 0
    task_availability = dict.fromkeys(task_counts.keys(), 0)

    while task_counts and any(task_counts.values()):
        next_tasks = task_counts.most_common(n+1) if n > 0 else task_counts.most_common()
        # set an idle time for next use  on the selected task
        for t, _ in next_tasks:
            if task_availability[t] <= interval:
                task_counts[t] -= 1
                if task_counts[t] == 0:
                    del task_counts[t]
                task_availability[t] = interval + n + 1
                break
        interval += 1
    return interval

def least_interval(tasks: List[str], n: int) -> int:
    # count number of each tasks
    task_counts = Counter(tasks)
    max_count = max(task_counts.values())
    max_count_tasks = sum(1 for count in task_counts.values() if count == max_count)

    part_count = max_count - 1
    part_length = n - (max_count_tasks - 1)
    empty_slots = part_count * part_length
    available_tasks = len(tasks) - max_count * max_count_tasks
    idles = max(0, empty_slots - available_tasks)

    return len(tasks) + idles
# debug your code below
print(least_interval(['A', 'A', 'B'], 2))

Merge Intervals

from typing import List

def mergeIntervals(intervals: List[List[int]]) -> List[List[int]]:
    if not intervals:
        return []

    intervals.sort(key=lambda x: x[0])

    result = [intervals[0]]

    for i in range(1, len(intervals)):
        current_interval = result[-1]
        next_interval = intervals[i]

        # If the curtrent interval overlaps with the next interval, merge
        if current_interval[1] >= next_interval[0]:
            current_interval[1] = max(current_interval[1], next_interval[1])
        else:
            # Otherwise, add the next interval to the result.
            result.append(next_interval)
    return result

# debug your code below
print(mergeIntervals([[1,3], [2,6], [8,10], [15,18]]))

Palindrome check

# Time Complexity: O(n)
# Space Complexity: O(n) for filtered characters and reversed characters
def is_palindrome(s: str) -> bool:
    # Filter out non-alphanumeric characters and convert to lowercase
    filtered_chars = [char.lower() for char in s if char.isalnum()]

    # Check if the filtered string reads the same forward and backward
    return filtered_chars == filtered_chars[::-1]

# Time Complexity: O(n)
# Space Complexity: O(1) for two pointers
def is_palindrome(s: str) -> bool:
    left, right = 0, len(s) - 1

    while left < right:
        while left < right and not s[left].isalnum():
            left += 1

        while left < right and not s[right].isalnum():
            right -= 1

        if s[left].lower() != s[right].lower():
            return False

        left += 1
        right -= 1

    return True

Remove Duplicates

def identify_adjacent(s: str, k: int) -> str:
    stack = [] # Store as [char, count]

    for char in s:
        if stack and stack[-1][0] == char:
            stack[-1][1] += 1 # increment consecutive
        else:
            stack.append([char, 1])

        if stack[-1][1] == k: # Check if the count has reached k
            stack.pop()
    return "".join(char * count for char, count in stack)

print(identify_adjacent("abcd", 2))

def has_good_subarray(nums, k):
    if nums is None:
        return False

    MIN_GOOD_SUBARRAY_LIMIT = 2

    sum_so_far = 0
    remainder_index_map = {0: -1}

    for current_index, num in enumerate(nums):
        sum_so_far += num
        remainder = sum_so_far % k

        if remainder not in remainder_index_map:
            remainder_index_map[remainder] = current_index
        elif current_index - remainder_index_map[remainder] >= MIN_GOOD_SUBARRAY_LIMIT:
            return True
    return False
print(has_good_subarray([23, 2, 4, 7], 6))
